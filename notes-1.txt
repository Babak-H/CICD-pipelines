def get_panel_name(url, dashboard, panel):
    if consts.IS_CLOUD:
        api_tokens = parse_yaml_file(consts.GRAFANA_API_KEYS_PATH)
        secret_api_key = get_secret(consts.SECRETS_NAME, consts.REGION_NAME)
        grafana_secrets_name = api_tokens[url]
        grf_obj = GrafanaApi(
            api_token=secret_api_key[grafana_secrets_name], base_url=url
        )
    else:
        api_tokens = parse_yaml_file(consts.GRAFANA_API_KEYS_PATH_LOCAL)
        grf_obj = GrafanaApi(api_token=api_tokens[url], base_url=url)
    dash_data = grf_obj.get_dashboard(uid=dashboard).json()
    panel_id = re.findall(r"viewPanel=(\d+)", panel)
    dashboard_name = dash_data.get("dashboard", {}).get("title", "Unknown_name")
    if panel_id:
        for panel in dash_data["dashboard"]["panels"]:
            if "panels" in panel:
                for child_panel in panel["panels"]:
                    if child_panel["id"] == int(panel_id[0]):
                        return dashboard_name + "-" + child_panel["title"]
                    elif panel["id"] == panel_id:
                        return dashboard_name + "-" + panel["title"]
        return dashboard_name + "panel_id_" + panel_id[0]
    return dashboard_name


def tab_handler(panel, new_dash_dict, dash_uid, hc, gs):
    try:
        try:
            panel_info = f"?{panel.split('?')[1]}"
        except IndexError:
            panel_info = ""
        range_filter = f"&from=now-{consts.TIME_RANGE}h&to=now"
        new_url = new_dash_dict[dash_uid]
        joined_url = new_url + panel_info
        if "orgId" not in joined_url:
            joined_url = joined_url + "?orgId=1"

        url_result = urllib.parse.urlparse(joined_url)
        panel_query = url_result.query
        window_height = hc.get("windows_height", consts.WINDOW_HEIGHT_DEFAULT)
        if panel_query.__contains__("from"):
            suffix = f"&{panel_query.split('&')[-2]}&{panel_query.split('&')[-1]}"
            updated_url = joined_url.removesuffix(suffix)
            dash_url_with_range = f"{updated_url}{range_filter}"
            gs.open_new_tab(dash_url_with_range, window_height)
        else:
            dash_url_with_range = f"{joined_url}{range_filter}"
            gs.open_new_tab(dash_url_with_range, window_height)
    except (KeyError, json.decoder.JSONDecodeError) as e:
        LOGGER.warning(f"f(Cannot capture panel {panel}), {e}")
    except Exception as e:
        LOGGER.critical(f"Unknown error, appeared during capture {e}")
        raise e


async def get_grafana_results(
    dash_list, grafana_api_url, health_checks, scenario_to_annotate
):
    get_metadata = dynamodb_service.get_item(
        table_name="soteria_metadata",
        primary_key={"scenario_id": scenario_to_annotate["scenario_id"]},
    )
    new_dash_dict = {}
    screenshot_names = []
    with Selenium(
        base_url=f"{grafana_api_url}/d/", screenshot_folder=consts.SCREENSHOT_FOLDER
    ) as gs:
        gs.graphana_auth()
        for dbd_list in dash_list:
            for key, val in dbd_list.items():
                new_dash_dict[key] = val["url"]
        for hc in health_checks:
            dash_uid_list = hc["dashboards"]
            for dash_uid, panels in dash_uid_list.items():
                for panel in panels:
                    LOGGER.debug(f"Starting Selenium for panel {panel}")
                    tab_handler(panel, new_dash_dict, dash_uid, hc, gs)
                    start_ = get_metadata["scenario_metadata"]["start"]
                    end_ = get_metadata["scenario_metadata"]["end"]
                    screenshot_names.append(
                        f"{get_panel_name(url=grafana_api_url, dashboard=dash_uid, panel=panel)}-"
                        f"S_{transform_time(start_)}-"
                        f"E_{transform_time(end_)}"
                    )
            time.sleep(consts.SCREENSHOT_DELAY)
            gs.get_screenshots(screenshot_names=screenshot_names)
        gs.driver_quit()


class Grafana:
    def __init__(self, grafana_api_url, api_token) -> None:
        self.logger = logging.getLogger(__name__)
        self.grafana_api_url = grafana_api_url
        self.ga = GrafanaApi(api_token=api_token, base_url=grafana_api_url)
        self.did_list: list = []
        self.annotation_id = 0

    def build_dashboard_json(self, conf, dashboard_dict, scenario_name, folder_id=0):
        try:
            seal_id = ""
            if conf == "product":
                seal_id = 105250
            elif conf == "infra":
                seal_id = ""
            time = datetime.now()
            d1 = time.strftime("%Y-%m-%d-%H:%M:%S")
            title = dashboard_dict["dashboard"]["title"]
            dash_mod_name = f"{d1}-{title}-{seal_id}-{conf}-Gremlin-{scenario_name}"
            dashboard_dict["meta"] = None
            dashboard_dict["dashboard"]["title"] = dash_mod_name
            dashboard_dict["dashboard"]["id"] = None
            dashboard_dict["dashboard"]["uid"] = None
            dash_payload = dashboard_dict["dashboard"]
            data = {
                "dashboard": dash_payload,
                "folderId": folder_id,
                "overwrite": False,
            }
            return json.dumps(data, indent=4)
        except Exception as e:
            self.logger.critical(f"Build Dashboard json failed, Error: {e}")
            raise e

    def get_dash_uid_by_name(self, dashboard_dict, folder_id):
        list_of_all_dashboard = self.ga.scan_folder(folder_id).json()
        for dash in list_of_all_dashboard:
            title = dash["title"]
            if title == dashboard_dict["dashboard"]["title"]:
                LOGGER.warning(
                    f"The dashboard was copied with an error, but the copy exists in Grafana. "
                    f"Please check collected screenshots for the {title}"
                )
                return dash
        else:
            LOGGER.critical(
                f"Cannot copy dashboard {dashboard_dict['dashboard']['title']}. Exiting"
            )
            exit()

    def generate_dashboard(self, folder_id, health_checks, conf, scenario_name):
        for dashboard in health_checks:
            dash_uid_list = dashboard["dashboards"].keys()
            for dash_uid in dash_uid_list:
                try:
                    res = self.ga.get_dashboard(dash_uid)
                    dashboard_dict = json.loads(res.text)
                    dashboard_input_json = self.build_dashboard_json(
                        conf, dashboard_dict, scenario_name, folder_id
                    )
                    res = self.ga.create_dashboard(dashboard_input_json)
                    if res.status_code != requests.codes.ok:
                        LOGGER.debug(
                            f"Grafana return {res.status_code}, fallback to "
                            "checking if the target dashboard copied with the error"
                        )
                        r = self.get_dash_uid_by_name(dashboard_dict, folder_id)
                    else:
                        r = json.loads(res.text)
                    custom = {dash_uid: r}
                    self.did_list.append(custom)
                except Exception as e:
                    self.logger.warning(
                        f"Creation of the dashboard failed, Error: {e}, skipping {dash_uid}"
                    )
        return self.did_list

    def create_annotation(self, run_time, ann_tags):
        try:
            get_metadata = dynamodb_service.get_item(
                table_name="soteria_metadata",
                primary_key={"scenario_id": run_time["scenario_id"]},
            )
            start_time = get_metadata["scenario_metadata"]["start"]
            end_time = get_metadata["scenario_metadata"]["end"]
            data = {
                "time": int(start_time),
                "timeEnd": int(end_time),
                "tags": ann_tags,
                "text": "Annotation for Soteria Chaos Automation",
            }
            input_data = json.dumps(data)
            re_annotation = self.ga.create_annotation(input_data)
            self.annotation_id = re_annotation.json()["id"]
        except Exception as e:
            self.logger.critical(f"Create annotation failed, Error: {e}")
            raise e

    def delete_annotation(self):
        try:
            self.ga.delete_annotation(self.annotation_id)
        except Exception as e:
            self.logger.critical(
                f"Cannot remove annotation {self.annotation_id}, Error: {e}"
            )

    def get_dashboard_uid(self, dash_list) -> list:
        try:
            dash_uid_list = [
                val["uid"] for dash_uid in dash_list for key, val in dash_uid.items()
            ]
            return dash_uid_list
        except Exception as e:
            self.logger.critical(f"Dashboard uid retrieval failed, Error: {e}")
            raise e

    def build_annotation_json(self, conf, gc, res_dict, dash_title):
        try:
            if re.search(conf, dash_title):
                input_conf = gc
                annotation_list = res_dict["dashboard"]["annotations"]["list"]
                tags = [hc["tags"] for hc in input_conf]
                if len(annotation_list) > 1:
                    for i in annotation_list:
                        if "builtIn" not in i:
                            i["tags"] = tags
                else:
                    input_dict = {
                        "datasource": "-- Grafana --",
                        "enable": True,
                        "hide": True,
                        "iconColor": "#FADE2A",
                        "limit": 100,
                        "matchAny": True,
                        "name": "soteria",
                        "showIn": 0,
                        "tags": tags,
                        "type": "tags",
                    }
                    annotation_list.append(input_dict)
        except Exception as e:
            self.logger.critical(f"Build annotation json failed, Error: {e}")
            raise e

    def inject_annotation(self, dash_list, folder_id, k, v):
        try:
            for dash_uid in self.get_dashboard_uid(dash_list):
                result = self.ga.get_dashboard(dash_uid)
                res_dict = json.loads(result.text)
                dash_title = res_dict["dashboard"]["title"]
                self.logger.debug("Creating annotation")
                self.build_annotation_json(k, v, res_dict, dash_title)
                data = {
                    "dashboard": res_dict["dashboard"],
                    "folderId": folder_id,
                    "overwrite": True,
                }
                dashboard_input_json = json.dumps(data, indent=4)
                self.ga.update_dashboard(dashboard_input_json)
        except Exception as e:
            self.logger.critical(f"Inject annotation failed, Error: {e}")
            raise e

    def cleanup_soteria_dashboards(self):
        for dashboard in self.did_list:
            try:
                dash_uid = next(iter(dashboard.values()))["uid"]
                self.ga.delete_dashboard(dash_uid)
            except Exception as e:
                self.logger.warning(f"Cannot remove dashboard {e}")

def upload_files_to_s3(
    attack_target,
    source_folder_path=consts.SCREENSHOT_FOLDER,
    bucket_name=consts.BUCKET_NAME,
    target_folder=None,
):
    """
    Uploads the contents of a directory to an S3 bucket, preserving the folder structure.

    :param attack_target:
    :param source_folder_path:
    :param bucket_name:
    :param target_folder:
    """
    # Create an S3 client
    s3 = boto3.client("s3")

    if target_folder is None:
        target_folder = (
            attack_target + "-" + datetime.datetime.now().strftime("%Y-%m-%d %H-%M")
        )
    # Walk through all the files and subdirectories in the local directory
    for root, dirs, files in os.walk(source_folder_path):
        for file_name in files:
            # Get the full path of the file
            file_path = os.path.join(root, file_name)

            # Construct the key for S3, by removing the base directory and preserving folder structure
            relative_path = os.path.relpath(file_path, source_folder_path)
            s3_key = os.path.join(target_folder, relative_path)

            try:
                # Upload the file to S3 bucket
                s3.upload_file(file_path, bucket_name, s3_key)
                LOGGER.info(f"File '{file_name}' uploaded to '{bucket_name}/{s3_key}'")
            except NoCredentialsError:
                LOGGER.critical("AWS credentials not available.")
                return
            except Exception as e:
                LOGGER.critical(f"Error uploading file '{file_name}': {str(e)}")
        return f"s3://{consts.BUCKET_NAME}/{s3_key.split('/')[0]}"

def services_to_skip_for_env(base_tf_path: str, env: str) -> list:
    with open(f"{base_tf_path}/{env}/terraform.tfvars") as fp:
        tf_vars = hcl2.load(fp)
        skipped_modules = tf_vars.get("skipped_modules", [])
        return skipped_modules


def services_to_skip_for_all_envs(base_tf_path: str) -> dict:
    env_map = {}
    env_list = list(os.listdir(base_tf_path))
    for env in env_list:
        services_to_skip = services_to_skip_for_env(base_tf_path, env)
        env_map[env.lower()] = services_to_skip
    return env_map


def create_env_map_file(base_tf_path: str, env_map_dir: str) -> dict:
    env_map = services_to_skip_for_all_envs(base_tf_path)
    Path(env_map_dir).mkdir(exist_ok=True)
    env_map_path = f"{env_map_dir}/{config.env_map_file}"
    logging.info(f"Creating env map file: {env_map_path}")
    with open(env_map_path, "w") as map_file:
        json.dump(env_map, map_file)
        return env_map


def filter_out_services_to_skip(services):
    with open(config.env_map_path) as map_file:
        env_map = json.load(map_file)
        services_to_skip = env_map[config.deploy_env]
        services_to_skip = [service.replace("_", "-") for service in services_to_skip]
        logging.info(
            f"environment: {config.deploy_env}, services_to_skip: {services_to_skip}",
        )
        services = [
            service for service in services if service["name"] not in services_to_skip
        ]
    return services

SERVICE_SUFFIXES = ["-alpha", "-bravo"]


class RequestedService:
    def __init__(self, name: str, permissions: tuple = ()):
        self.name = name
        self.permissions = permissions

    def __eq__(self, other):
        return self.name == other.name and self.permissions == other.permissions


def load_services() -> list[RequestedService]:
    with open(config.services_map_path) as f:
        services = yaml.load(f, yaml.SafeLoader)

    services = filter_out_services_to_skip(services)
    logging.info(
        f"Configuring {len(services)} active services: {[s['name'] for s in services]}",
    )

    return [*map(lambda x: RequestedService(*x.values()), services)]


def get_sa_names_for_service(service_name: str) -> set[str]:
    return {service_name + service_suffix for service_suffix in SERVICE_SUFFIXES}


def update_tm_from_req(
    services_req: list[RequestedService],
    sa_names: set[str],
) -> tuple[Iterable[RequestedService], Iterable[RequestedService], set[str]]:
    service_new = list()
    service_exst = list()

    for service_req in services_req:
        sa_req_names = get_sa_names_for_service(service_req.name)

        if sa_req_names.issubset(sa_names):
            service_exst.append(service_req)
        else:
            service_new.append(service_req)

        sa_names = sa_names.difference(sa_req_names)

    return service_new, service_exst, sa_names


class Runner:
    def __init__(self):
        self.secret_manager = SecretManager(
            config.region_name,
            config.seal_id,
            config.deploy_id,
            config.deploy_env,
        )
        self.vault_controller = VaultController()

    def run(self):
        logging.info("Starting... ")
        logging.info(f"threshold_days={config.threshold_days} ")
        logging.info(f"env={config.deploy_env} ")
        logging.info(
            f"delete_old_accounts={config.delete_old_accounts} ",
        )  # from TM (freeze)
        logging.info(f"create_new_accounts={config.create_new_accounts} ")  # in TM
        logging.info(f"rotate_account_tokens={config.rotate_accounts_tokens}")  # in SM
        logging.info(f"rotate_now={config.rotate_now}")  # in TM then SM

        # Initialization
        rotator_token = self.secret_manager.fetch_secret(
            config.rotator_service_name,
        ).fetch_token()

        logging.debug("Rotator token: %s", rotator_token)

        vault = self.vault_controller.init(rotator_token)

        logging.debug(f"Vault accounts: {self.vault_controller}")

        services_req = load_services()

        service_new, service_exst, sa_del = update_tm_from_req(
            services_req,
            vault.get_active_service_account_names(),
        )

        def create_new_service(svc: RequestedService):
            sa_names = sorted(get_sa_names_for_service(svc.name))
            ids, tokens = self.vault_controller.create_sa(sa_names, svc.permissions)
            self.secret_manager.save_secret(svc.name, ids, tokens)

        for service in service_new:
            logging.info("Adding %s%s", "-" * 8, service.name)
            if not config.create_new_accounts:
                logging.info(
                    "Skipping creation %s since create_new_accounts=%s",
                    service.name,
                    config.create_new_accounts,
                )
                continue

            create_new_service(service)

        for service in service_exst:
            logging.info("Updating %s%s", "-" * 8, service.name)

            sa_names_exst = get_sa_names_for_service(service.name)

            self.vault_controller.update_permissions(sa_names_exst, service.permissions)

            sm_secret = self.secret_manager.fetch_secret(service.name)

            # Fixing existing secrets if they are invalid
            if not sm_secret.valid() or config.rotate_now:
                create_new_service(service)
                continue

            if config.rotate_accounts_tokens:
                if self.secret_manager.should_refresh(sm_secret):
                    new_token = self.vault_controller.refresh_token(
                        sm_secret.token_last.sa_id,
                    )
                    self.secret_manager.rotate_secret(sm_secret, new_token)

        for sa_name in sa_del:
            logging.info("Removing %s%s", "-" * 8, sa_name)

            if not config.delete_old_accounts:
                logging.info(
                    "Skipping deletion %s since delete_old_accounts=%s",
                    sa_name,
                    config.delete_old_accounts,
                )
                continue

            self.vault_controller.remove_sa(sa_name)



import datetime
import json
import logging
from collections import defaultdict

import boto3

class SecretToken:
    def __init__(self, sa_id=None, token=None):
        self.sa_id = sa_id
        self.token = token
        self.last_updated = 0 if token is None else datetime.datetime.now().timestamp()


class Secret:
    def __init__(self, arn_name: str, secret_tokens: list[SecretToken]):
        self.arn_name = arn_name
        self.secret_tokens: list[SecretToken] = secret_tokens

    @property
    def token_1(self) -> SecretToken:
        return self.secret_tokens[0]

    @property
    def token_last(self) -> SecretToken:
        return self.secret_tokens[1]

    def valid(self) -> bool:
        return len(self.secret_tokens) == 2

    def fetch_token(self):
        return self.token_1.token

    @staticmethod
    def from_json_response(secret_arn: str, json_response: dict[str, str]):
        raw_secret_tokens: dict[int, SecretToken] = defaultdict(SecretToken)

        if len(json_response) == 1:
            # This is the intial rotator service account - just single value - converting...
            raw_secret_tokens[0] = SecretToken(
                sa_id=-1,
                token=json_response.popitem()[1],
            )
        else:
            for key, val in json_response.items():
                import re

                PATTERN = r"token(\d)_?(.*)"
                m = re.fullmatch(PATTERN, key)

                if m is None:
                    logging.warning(
                        "Cannot understand the pattern in the secret %s: %s",
                        secret_arn,
                        key,
                    )
                    continue

                tkn_id = int(m.group(1))
                tkn_type = m.group(2)

                if not tkn_type:
                    raw_secret_tokens[tkn_id].token = val
                elif tkn_type == "id":
                    raw_secret_tokens[tkn_id].sa_id = val
                elif tkn_type == "last_updated":
                    raw_secret_tokens[tkn_id].last_updated = float(val)

        secret_tokens: list[SecretToken] = [
            raw_secret_tokens[tkn_id]
            for tkn_id in sorted(raw_secret_tokens.keys())
            if None
            not in (raw_secret_tokens[tkn_id].sa_id, raw_secret_tokens[tkn_id].token)
        ]
        return Secret(secret_arn, secret_tokens)

    def serialize(self):
        if not self.valid():
            raise SecretsManagerError(f"Secret {self.arn_name} has wrong values.")

        secret_values = {}

        for idx, secret_token in enumerate(self.secret_tokens):
            secret_values[f"token{idx + 1}"] = secret_token.token
            secret_values[f"token{idx + 1}_id"] = secret_token.sa_id
            secret_values[f"token{idx + 1}_last_updated"] = secret_token.last_updated

        return secret_values


class SecretManager:
    def __init__(self, region_name: str, seal_id: str, deploy_id: str, deploy_env: str):
        self.client = boto3.client(
            service_name="secretsmanager",
            region_name=region_name,
        )
        self.seal_id = seal_id
        self.deploy_id = deploy_id
        self.deploy_env = deploy_env

    def build_secret_name(self, service_name: str) -> str:
        return f"{self.seal_id}-{self.deploy_id}-{self.deploy_env}-{service_name}-tennessee-token"

    def __load_secret(self, secret_arn: str) -> dict[str, str]:
        logging.debug("Loading secret: %s", secret_arn)
        try:
            response = self.client.get_secret_value(SecretId=secret_arn)
            value = response["SecretString"]
        except self.client.exceptions.ResourceNotFoundException:
            logging.warning("AWS secret %s doesn't have value", secret_arn)
            value = "{}"

        logging.debug("Secret successfully extracted from %s", secret_arn)
        return json.loads(value)

    def fetch_secret(self, service_name: str) -> Secret:
        secret_arn = self.build_secret_name(service_name)
        json_response = self.__load_secret(secret_arn)

        return Secret.from_json_response(secret_arn, json_response)

    def save_secret(self, service_name, ids, tokens) -> None:
        arn_name = self.build_secret_name(service_name)
        secret_tokens = [
            SecretToken(sa_id=sa_id, token=token) for (sa_id, token) in zip(ids, tokens)
        ]
        sm_secret = Secret(arn_name, secret_tokens=secret_tokens)
        self.__save_secret(sm_secret)

    @staticmethod
    def should_refresh(sm_secret: Secret) -> bool:
        last_updated = sm_secret.token_1.last_updated
        age = datetime.timedelta(
            seconds=datetime.datetime.now().timestamp() - last_updated,
        )
        logging.debug(
            f"Secret {sm_secret.arn_name} ID {sm_secret.token_1.sa_id} has token age '{age}'",
        )
        return age > datetime.timedelta(days=threshold_days) / 2

    def rotate_secret(self, sm_secret: Secret, token: str):
        prev_last_token = sm_secret.token_last
        sm_secret.secret_tokens = [
            SecretToken(prev_last_token.sa_id, token),
        ] + sm_secret.secret_tokens[:-1]
        self.__save_secret(sm_secret)

    def __save_secret(self, sm_secret: Secret):
        secret_values = sm_secret.serialize()

        self.client.put_secret_value(
            SecretId=sm_secret.arn_name,
            SecretString=json.dumps(secret_values),
        )
        logging.info("Saving secret: %s", sm_secret.arn_name)
