import hvac
import requests
from botocore.exceptions import ClientError
from requests.adapters import HTTPAdapter

# Environment variables
AWS_REGION = os.getenv("AWS_REGION", "eu-west-1")
S3_BUCKET_HAULT_TOKEN = os.getenv("S3_BUCKET_NAME_HAULT_TOKEN")
S3_OBJECT_HAULT_TOKEN = os.getenv("S3_OBJECT_NAME_HAULT_TOKEN")
AWS_ROOT_CA = os.getenv("AWS_ROOT_CA", "/tls")
HAULT_URL = "https://your-vault-url"
SPECIFIC_SECRET_PATH = "your/specific/secret/path"

def get_session():
    """Function to setup retry and pass it as session as vault is run into local pod and"""
    adapter = HTTPAdapter(
        max_retries=Retry(
            total=3,
            backoff_factor=0.1,
            status_forcelist=[412, 500, 502, 503],
            raise_on_status=False,
        )
    )
    session = requests.Session()
    session.verify = f"{AWS_ROOT_CA}/root_ca.pem"
    session.mount("https://", adapter)
    return session

def retrieve_s3_object():
    """Function to connect to AWS S3 and retrieve an object"""

    # Initialize the S3 client
    s3_client = boto3.client("s3", region_name=AWS_REGION)

    try:
        # Retrieve the object from the S3 bucket
        response = s3_client.get_object(Bucket=S3_BUCKET_HAULT_TOKEN, Key=S3_OBJECT_HAULT_TOKEN)
        
        # Read the content of the object
        object_data = response['Body'].read().decode('utf-8')
        
        print("Successfully retrieved the object from S3.")
        return object_data
    except ClientError as err:
        print(f"Failed to retrieve the object: {err}")
        return None

def hault_collector():
    """Function to retrieve a specific secret token from Hashicorp Vault"""

    # Retrieve the token from S3
    service_account_token = retrieve_s3_object()
    if not service_account_token:
        print("Failed to retrieve the service account token from S3.")
        return None

    # Initialize Hashicorp Vault access
    hault_client = hvac.Client(url=HAULT_URL, session=get_session())

    # Authenticate with Vault using the service account token
    hault_client.auth_kubernetes(role="vault-monitor-role", jwt=service_account_token)

    if not hault_client.is_authenticated():
        print("Authentication failed")
        return None

    try:
        # Retrieve the specific secret token
        secret_data = hault_client.secrets.kv.v1.read_secret(path=SPECIFIC_SECRET_PATH)["data"]
        specific_secret_token = secret_data.get("your_secret_key")  # Replace with the actual key
        print("Retrieved Secret Token:", specific_secret_token)
        return specific_secret_token
    except hvac.exceptions.InvalidPath:
        print("Invalid path for the secret")
        return None
    except hvac.exceptions.VaultError as error_hvac:
        print("Vault error:", error_hvac)
        return None

# Example usage
if __name__ == "__main__":
    secret_token = hault_collector()
    if secret_token:
        print("Successfully retrieved the secret token.")
    else:
        print("Failed to retrieve the secret token.")

To add a token to a Kubernetes ServiceAccount, you typically create a Secret that contains the token and then associate it with the ServiceAccount. Here's a step-by-step guide on how to do this:

Create a Secret with the Token:

First, you need to create a Kubernetes Secret that contains your token. You can do this by creating a YAML file for the secret or using the kubectl command directly.

Here's an example of creating a secret using a YAML file:

yaml



apiVersion: v1
kind: Secret
metadata:
  name: my-token-secret
type: Opaque
data:
  token: <base64-encoded-token>
Replace <base64-encoded-token> with your token encoded in base64. You can encode your token using a command like:

bash



echo -n "your-token" | base64
Apply the secret using:

bash



kubectl apply -f secret.yaml
Alternatively, you can create the secret directly using kubectl:

bash



kubectl create secret generic my-token-secret --from-literal=token=your-token
Associate the Secret with a ServiceAccount:

Once you have the secret, you need to associate it with a ServiceAccount. You can do this by modifying the ServiceAccount to reference the secret.

Here's an example of how to modify a ServiceAccount:

yaml



apiVersion: v1
kind: ServiceAccount
metadata:
  name: my-service-account
secrets:
  - name: my-token-secret
Apply the changes using:

bash



kubectl apply -f serviceaccount.yaml
Verify the Association:

You can verify that the secret is associated with the ServiceAccount by describing the ServiceAccount:

bash



kubectl describe serviceaccount my-service-account
You should see the secret listed under the Mountable secrets or Image pull secrets section, depending on how it's used.

By following these steps, you can add a token to a Kubernetes ServiceAccount by creating a secret and associating it with the ServiceAccount.
